# LogGYM - comandos uteis para DEV
# Ambiente alvo: Windows / PowerShell

==================================================
1. Instalacao inicial
==================================================

npm install

==================================================
2. Rodar o projeto em DEV
==================================================

# Terminal 1 - subir o Metro
npm run start

# Terminal 1 - subir o Metro limpando cache
npm run start:reset

# Terminal 2 - instalar e rodar no Android em debug
npm run android:dev

# Corrigir conflito de versao do adb no Windows antes de rodar
npm run adb:fix

# Alternativa direta do React Native CLI
npm run android

# Painel web React
npm run web:dev

==================================================
3. Validacao de codigo
==================================================

# TypeScript
npm run typecheck

# ESLint
npm run lint

# ESLint com autofix
npm run lint:fix

# Verificacao completa do projeto
npm run verify

# Jest
npm test

# Lint do painel web
npm run web:lint

==================================================
4. Build APK para testes
==================================================

# Gerar APK debug
npm run apk:debug

# Gerar APK release
npm run apk:release

# Validar se a assinatura release esta pronta antes do build
npm run release:check

# Validar se o Firebase esta configurado para mobile + web
npm run firebase:doctor

# Instalar manualmente o APK debug no dispositivo
C:\Users\Samuel\AppData\Local\Android\Sdk\platform-tools\adb install -r .\android\app\build\outputs\apk\debug\app-debug.apk

==================================================
5. ADB e dispositivo
==================================================

# Diagnosticar conflito de versao do adb
where.exe adb

# Corrigir e validar o adb usado pelo projeto
npm run adb:fix

# Se aparecer "Minimal ADB and Fastboot", remova
# C:\Program Files (x86)\Minimal ADB and Fastboot
# do PATH do Windows ou desinstale esse pacote.

# Listar dispositivos conectados
npm run adb:fix

# Se quiser usar diretamente o adb correto do SDK
C:\Users\Samuel\AppData\Local\Android\Sdk\platform-tools\adb devices

# Reiniciar o servidor ADB
C:\Users\Samuel\AppData\Local\Android\Sdk\platform-tools\adb kill-server
C:\Users\Samuel\AppData\Local\Android\Sdk\platform-tools\adb start-server

# Fazer reverse da porta do Metro para dispositivo fisico
C:\Users\Samuel\AppData\Local\Android\Sdk\platform-tools\adb reverse tcp:8081 tcp:8081

# Ver logs do app e do React Native
C:\Users\Samuel\AppData\Local\Android\Sdk\platform-tools\adb logcat *:S ReactNative:V ReactNativeJS:V

# Limpar dados do app instalado
C:\Users\Samuel\AppData\Local\Android\Sdk\platform-tools\adb shell pm clear com.loggym

# Desinstalar o app
C:\Users\Samuel\AppData\Local\Android\Sdk\platform-tools\adb uninstall com.loggym

==================================================
6. Limpeza quando der problema de cache/build
==================================================

# Limpar build Android
cd .\android
.\gradlew.bat clean
cd ..

# Rodar Metro com cache zerado
npm run start:reset

# Reinstalar dependencias do zero
Remove-Item -LiteralPath .\node_modules -Recurse -Force
Remove-Item -LiteralPath .\package-lock.json -Force
npm install

# Limpar caches comuns do Metro/Watchman no Windows
Remove-Item -LiteralPath .\node_modules\.cache -Recurse -Force -ErrorAction SilentlyContinue
npm run start:reset

==================================================
11. Release assinada
==================================================

# 1. Gerar a keystore de release
keytool -genkeypair -v -storetype PKCS12 -keystore ".\android\keystores\loggym-upload.jks" -alias loggym-upload -keyalg RSA -keysize 2048 -validity 9125

# 2. Abrir o modelo e usar como base
notepad .\android\release-signing.example.properties

# 3. Adicionar as chaves LOGGYM_* ao gradle.properties do usuario
notepad "$env:USERPROFILE\.gradle\gradle.properties"

# 4. Validar a configuracao release
npm run release:check

# 5. Gerar a APK release assinada
npm run apk:release
